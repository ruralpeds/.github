// ===== COLOR & STYLE MAPS ===== const typeColors = { assess: { fill: '#1e3a5f', stroke: '#3b82f6', badge: '#3b82f6', label: 'Assessment' }, decide: { fill: '#3b1f5e', stroke: '#8b5cf6', badge: '#8b5cf6', label: 'Decision' }, treat: { fill: '#1a3d2e', stroke: '#10b981', badge: '#10b981', label: 'Treatment' }, escalate: { fill: '#3d2e1a', stroke: '#f59e0b', badge: '#f59e0b', label: 'Escalation' }, wean: { fill: '#1a3d3d', stroke: '#06b6d4', badge: '#06b6d4', label: 'Weaning' }, stop: { fill: '#1a2d1a', stroke: '#22c55e', badge: '#22c55e', label: 'Discharge' }, adjunct: { fill: '#2e1a3d', stroke: '#f97316', badge: '#f97316', label: 'Adjunctive' }, }; // ===== D3 TREE SETUP ===== const width = window.innerWidth; const height = window.innerHeight; const nodeW = 210, nodeH = 52; const svg = d3.select('#tree-svg') .attr('width', width) .attr('height', height); const g = svg.append('g'); const zoom = d3.zoom() .scaleExtent([0.15, 2.5]) .on('zoom', (e) => g.attr('transform', e.transform)); svg.call(zoom); const treemap = d3.tree().nodeSize([nodeW + 30, nodeH + 90]); let root = d3.hierarchy(TREE_DATA, d => d.children); root.x0 = 0; root.y0 = 0; // Start with depth 2 collapsed root.descendants().forEach((d) => { if (d.depth > 2 && d.children) { d._children = d.children; d.children = null; } }); const linkGroup = g.append('g').attr('class', 'links'); const nodeGroup = g.append('g').attr('class', 'nodes'); function update(source) { const treeLayout = treemap(root); const nodes = treeLayout.descendants(); const links = treeLayout.links(); // Swap x/y for horizontal layout nodes.forEach(d => { const tmp = d.x; d.x = d.y * 1.4; d.y = tmp; }); // ===== LINKS ===== const link = linkGroup.selectAll('.link-group') .data(links, d => d.target.data.name); const linkEnter = link.enter().append('g').attr('class', 'link-group'); linkEnter.append('path') .attr('class', 'link-path') .attr('d', () => { const o = { x: source.x0 || 0, y: source.y0 || 0 }; return diagonal(o, o); }) .attr('stroke', d => typeColors[d.target.data.type]?.stroke || '#3b82f6'); const linkUpdate = linkEnter.merge(link); linkUpdate.select('.link-path') .transition().duration(500) .attr('d', d => diagonal(d.source, d.target)) .attr('stroke', d => typeColors[d.target.data.type]?.stroke || '#3b82f6'); link.exit().select('.link-path') .transition().duration(300) .attr('d', () => { const o = { x: source.x, y: source.y }; return diagonal(o, o); }) .remove(); link.exit().transition().duration(300).remove(); // ===== NODES ===== const node = nodeGroup.selectAll('.node-group') .data(nodes, d => d.data.name); const nodeEnter = node.enter().append('g') .attr('class', 'node-group') .attr('transform', `translate(${source.x0 || 0},${source.y0 || 0})`) .on('click', (e, d) => { e.stopPropagation(); if (e.shiftKey || e.metaKey) { showDetail(d.data); return; } if (d.children) { d._children = d.children; d.children = null; } else if (d._children) { d.children = d._children; d._children = null; } showDetail(d.data); update(d); }); // Node rectangle nodeEnter.append('rect') .attr('class', 'node-rect') .attr('x', -nodeW / 2) .attr('y', -nodeH / 2) .attr('width', nodeW) .attr('height', nodeH) .attr('fill', d => typeColors[d.data.type]?.fill || '#1a2035') .attr('stroke', d => typeColors[d.data.type]?.stroke || '#3b82f6'); // Type badge nodeEnter.append('rect') .attr('x', -nodeW / 2 + 6) .attr('y', -nodeH / 2 + 5) .attr('width', 50) .attr('height', 12) .attr('rx', 2) .attr('fill', d => typeColors[d.data.type]?.badge || '#3b82f6') .attr('opacity', 0.2); nodeEnter.append('text') .attr('class', 'node-badge') .attr('x', -nodeW / 2 + 8) .attr('y', -nodeH / 2 + 14) .text(d => typeColors[d.data.type]?.label?.toUpperCase() || ''); // Name label nodeEnter.append('text') .attr('class', 'node-label') .attr('x', -nodeW / 2 + 8) .attr('y', 2) .text(d => truncate(d.data.name, 28)); // Sublabel nodeEnter.append('text') .attr('class', 'node-sublabel') .attr('x', -nodeW / 2 + 8) .attr('y', 16) .text(d => truncate(d.data.sub || '', 34)); // Expand/collapse indicator nodeEnter.append('text') .attr('class', 'expand-icon') .attr('x', nodeW / 2 - 14) .attr('y', 4) .text(d => (d.children || d._children) ? (d.children ? '−' : '+') : ''); // UPDATE const nodeUpdate = nodeEnter.merge(node); nodeUpdate.transition().duration(500) .attr('transform', d => `translate(${d.x},${d.y})`); nodeUpdate.select('.expand-icon') .text(d => (d.children || d._children) ? (d.children ? '−' : '+') : ''); // EXIT node.exit().transition().duration(300) .attr('transform', `translate(${source.x},${source.y})`) .style('opacity', 0) .remove(); // Store positions nodes.forEach(d => { d.x0 = d.x; d.y0 = d.y; }); } function diagonal(s, d) { const mx = (s.x + d.x) / 2; return `M ${s.x},${s.y} C ${mx},${s.y} ${mx},${d.y} ${d.x},${d.y}`; } function truncate(str, n) { return str.length > n ? str.substr(0, n - 1) + '…' : str; } // ===== DETAIL PANEL ===== function showDetail(data) { const panel = document.getElementById('detail-panel'); const colors = typeColors[data.type] || typeColors.assess; document.getElementById('detail-type-badge').textContent = colors.label; document.getElementById('detail-type-badge').style.background = colors.badge + '22'; document.getElementById('detail-type-badge').style.color = colors.badge; document.getElementById('detail-title').textContent = data.name; document.getElementById('detail-body').innerHTML = data.detail || 'No additional details.'; panel.classList.add('visible'); } function closeDetail() { document.getElementById('detail-panel').classList.remove('visible'); } // ===== CONTROLS ===== function zoomIn() { svg.transition().duration(300).call(zoom.scaleBy, 1.3); } function zoomOut() { svg.transition().duration(300).call(zoom.scaleBy, 0.7); } function resetView() { svg.transition().duration(500).call( zoom.transform, d3.zoomIdentity.translate(180, height / 2).scale(0.65) ); } function expandAll() { root.descendants().forEach(d => { if (d._children) { d.children = d._children; d._children = null; } }); update(root); } function collapseAll() { root.descendants().forEach(d => { if (d.depth > 1 && d.children) { d._children = d.children; d.children = null; } }); update(root); } // ===== INIT ===== update(root); resetView(); // Close detail on background click svg.on('click', () => closeDetail()); // Resize handler window.addEventListener('resize', () => { const w = window.innerWidth; const h = window.innerHeight; svg.attr('width', w).attr('height', h); });
